# Backend
cd backend
python3 -m venv venv && source venv/bin/activate
pip install -r requirements.txt
python manage.py check --deploy   # you'll see the SECRET_KEY warning until you set a real one
python manage.py collectstatic --noinput

# Frontend
cd frontend
npm install
npx tsc --noEmit
npm run build     # requires real internet access to fonts.googleapis.com
